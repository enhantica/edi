cmake_minimum_required(VERSION 3.24)
project(assistant_native_contract_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(EDI_BUILD_APP OFF CACHE BOOL "" FORCE)
set(EDI_BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(EDI_BUILD_ASSISTANT_NATIVE ON CACHE BOOL "" FORCE)
get_filename_component(EDI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network)
add_subdirectory("${EDI_SOURCE}" edi)
add_executable(assistant_native_contract_probe probe.cpp)
target_link_libraries(assistant_native_contract_probe PRIVATE edi_assistant_native Qt6::Core Qt6::Network)
add_executable(assistant_qt_transport_probe qt_transport_probe.cpp)
target_link_libraries(assistant_qt_transport_probe PRIVATE edi_assistant_native Qt6::Core Qt6::Network)
add_executable(assistant_store_contract_probe store_probe.cpp)
target_link_libraries(assistant_store_contract_probe PRIVATE edi_assistant_native Qt6::Core Qt6::Network)
